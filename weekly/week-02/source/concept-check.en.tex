\parttitle{1. Concept Check}{Choose the single best answer for each question.}
\question{1}{You want to predict house prices using size, room count, and house age. Which description is most appropriate?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item Copying the target price three times
\item Necessarily keeping only one of the three features
\item Removing the cost function
\item Representing one house using multiple input features
\end{enumerate}\vfill
\question{2}{In $\hat y=w_1x_1+w_2x_2+b$, what does each weight do?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item Controls how much its feature contributes to the prediction
\item Determines the name of its feature
\item Determines only the number of data rows
\item Changes the actual target arbitrarily
\end{enumerate}\vfill
\question{3}{If $x=(2,3)$, $w=(4,1)$, and $b=1$, what is $\hat y=w^Tx+b$?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item $10$
\item $11$
\item $12$
\item $14$
\end{enumerate}\vfill
\question{4}{What is the main reason to use vectorization?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item To train on different targets
\item To express the same calculation concisely and efficiently with array operations
\item To replace every feature with one constant
\item To guarantee zero model error
\end{enumerate}\vfill
\question{5}{House size is measured in thousands, while room count is a single digit. Why apply feature scaling before gradient descent?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item To make all actual house prices equal
\item To remove features with small values
\item To make size and room count mean the same thing
\item To reduce differences in numerical ranges and help training progress smoothly
\end{enumerate}\vfill

\question{6}{A feature has mean $10$ and standard deviation $2$. What is the standardized value of $14$ using $(x-\mu)/\sigma$?}

\item $4$
\item $7$

\question{7}{You standardize a new house for prediction. Which mean and standard deviation should you use?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item Values calculated from only the new house
\item New random values at each prediction
\item The saved values calculated from the training data
\item The mean and standard deviation of target prices
\end{enumerate}\vfill
\question{8}{During training, the cost keeps increasing and oscillating strongly. What should you try first?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item Increase the learning rate further
\item Reduce the learning rate and monitor the cost
\item Replace all actual targets with zero
\item Stop recording the cost
\end{enumerate}\vfill
\question{9}{Price varies with house size $x$ along a curve. Which approach from this lesson could you try?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item Replace every input with 1
\item Keep every weight fixed at 0
\item Randomly change the target for the same input
\item Add a new feature such as $x^2$
\end{enumerate}\vfill
\question{10}{Why can $\hat y=w_1x+w_2x^2+b$ be treated as linear regression even though it is curved with respect to $x$?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item The parameters $w_1,w_2,b$ enter linearly
\item Every curve is a straight line
\item Inputs must be 0 or 1
\item The weights are not learned
\end{enumerate}\vfill
