\question{6}{For an example whose actual target is 1, the model predicts a positive probability of 0.01. Its logistic loss is greater than which alternative?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item Predicting a probability of 0.001
\item Predicting a probability of 0.9
\item Predicting the same probability of 0.01

\question{7}{A model fits its training data very well but often fails on unseen data. What should you suspect?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item Necessarily perfect generalization
\item A file format that stores too little data
\item Proof that classification is impossible
\item Overfitting
\end{enumerate}\vfill
\question{8}{How can regularization help reduce overfitting?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item By penalizing excessively large weights and discouraging overly complex fitting
\item By deleting all training targets
\item By rewarding predictions with large errors
\item By making every model perfect
\end{enumerate}\vfill
\question{9}{With other conditions fixed, what can happen if the regularization strength $\lambda$ is too large?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item Every error must disappear
\item The number of input features automatically increases
\item Weights can be suppressed so strongly that useful patterns are missed
\item The number of actual targets increases
\end{enumerate}\vfill
\question{10}{What is the best way to judge whether a classifier has learned successfully?}
\begin{enumerate}[label=(\arabic*),leftmargin=9mm,itemsep=0pt,topsep=2pt,parsep=0pt]
\item Look only at training accuracy
\item Also evaluate performance on data not used for training
\item Always prefer larger weights
\item Change every prediction to the same class
\end{enumerate}\vfill
